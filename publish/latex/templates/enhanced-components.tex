% 代码块环境（带标题）
\lstnewenvironment{smartcode}[2][]{
    \lstset{
        style=#2,
        title={#1},
        #1
    }
}{}

\lstnewenvironment{terminal}[1][]{
    \lstset{
        style=terminal,
        title={终端命令},
        #1
    }
}{}

\lstnewenvironment{configfile}[2][]{
    \lstset{
        style=config,
        title={配置文件：#2},
        #1
    }
}{}
