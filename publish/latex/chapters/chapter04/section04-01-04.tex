\section{HTML5 API与应用}\label{html5-apiux4e0eux5e94ux7528}

HTML5不仅带来了新的标签和元素，还引入了丰富的API，使Web应用具备了更强大的能力。这些API为智慧水利平台的开发提供了众多实用功能，从地理位置获取到本地存储，从离线应用到实时通信，极大拓展了Web应用的边界。

\subsection{1. 地理位置API
(Geolocation)}\label{ux5730ux7406ux4f4dux7f6eapi-geolocation}

地理位置API允许网页获取用户的地理位置信息，这对于水利现场作业、灾情上报和移动监测具有重要意义。

\subsubsection{1.1 基本用法}\label{ux57faux672cux7528ux6cd5}

\begin{lstlisting}
// 检查浏览器是否支持地理位置API
if (navigator.geolocation) {
    // 获取当前位置
    navigator.geolocation.getCurrentPosition(
        // 成功回调
        function(position) {
            const latitude = position.coords.latitude;
            const longitude = position.coords.longitude;
            const accuracy = position.coords.accuracy;
            
            console.log(`当前位置: 纬度 ${latitude}, 经度 ${longitude}, 精度 ${accuracy}米`);
            
            // 在地图上标记位置
            markOnMap(latitude, longitude);
            
            // 查找附近的水利设施
            findNearbyWaterFacilities(latitude, longitude);
        },
        // 错误回调
        function(error) {
            switch(error.code) {
                case error.PERMISSION_DENIED:
                    console.error("用户拒绝了位置请求");
                    break;
                case error.POSITION_UNAVAILABLE:
                    console.error("位置信息不可用");
                    break;
                case error.TIMEOUT:
                    console.error("获取位置超时");
                    break;
                case error.UNKNOWN_ERROR:
                    console.error("未知错误");
                    break;
            }
        },
        // 选项
        {
            enableHighAccuracy: true, // 尝试获取最高精度
            timeout: 10000,           // 超时时间：10秒
            maximumAge: 60000         // 位置缓存时间：1分钟
        }
    );
} else {
    console.error("浏览器不支持地理位置API");
}
\end{lstlisting}

\subsection{2. Web存储 (Web Storage)}\label{webux5b58ux50a8-web-storage}

Web Storage
API提供了在客户端存储数据的机制，比Cookie更安全、容量更大。在智慧水利平台中，可用于保存用户首选项、缓存监测数据和断网时的临时存储。

    console.log('旧值:', event.oldValue);

        applyTheme(event.newValue);
    } else if (event.key === 'dataRefreshInterval') {
        updateRefreshInterval(parseInt(event.newValue));
    }
});

\subsection{3. 数据通信API}\label{ux6570ux636eux901aux4fe1api}

HTML5提供了多种数据通信API，使Web应用能够与服务器进行实时双向通信，这对于监测数据实时更新至关重要。

\subsubsection{3.1 WebSocket}\label{websocket}

WebSocket是一种在单个TCP连接上进行全双工通信的协议，特别适合水利监测平台的实时数据推送。

\begin{lstlisting}
// 创建WebSocket连接
const socket = new WebSocket('wss://water-monitoring.example.com/realtime');

// 连接建立时触发
socket.addEventListener('open', function(event) {
    console.log('WebSocket连接已建立');
    
    // 订阅特定站点的数据
    socket.send(JSON.stringify({
        action: 'subscribe',
        stationIds: ['ST001', 'ST002', 'ST003'],
        dataTypes: ['waterLevel', 'flowRate', 'precipitation']
    }));
});

// 接收消息
socket.addEventListener('message', function(event) {
    const data = JSON.parse(event.data);
    console.log('从服务器接收到数据:', data);
    
    if (data.type === 'reading') {
        // 更新监测读数
        updateReading(data.stationId, data.dataType, data.value, data.timestamp);
    } else if (data.type === 'alert') {
        // 显示警报
        showAlert(data.alertLevel, data.message, data.stationId);
    }
});

// 错误处理
socket.addEventListener('error', function(event) {
    console.error('WebSocket错误:', event);
});

// 连接关闭
socket.addEventListener('close', function(event) {
    console.log('WebSocket连接已关闭', event.code, event.reason);
    
    // 尝试重新连接
    setTimeout(function() {
        console.log('尝试重新连接...');
        // 重新创建WebSocket连接
    }, 5000);
});

// 发送数据给服务器
function sendCommand(command, params) {
    if (socket.readyState === WebSocket.OPEN) {
        socket.send(JSON.stringify({
            command: command,
            params: params
        }));
    } else {
        console.error('WebSocket连接未开启');
    }
}
\end{lstlisting}

\subsubsection{3.2 Server-Sent Events
(SSE)}\label{server-sent-events-sse}

SSE是一种服务器向浏览器推送消息的技术，适合于单向数据流，如监测数据更新、警报通知等。

\begin{lstlisting}
// 创建EventSource连接
const evtSource = new EventSource('https://water-monitoring.example.com/events');

// 接收消息
evtSource.addEventListener('message', function(event) {
    const data = JSON.parse(event.data);
    console.log('接收到事件:', data);
    
    // 处理数据
    processEventData(data);
});

// 接收特定类型的事件
evtSource.addEventListener('water-level', function(event) {
    const data = JSON.parse(event.data);
    console.log('水位更新:', data);
    
    // 更新水位显示
    updateWaterLevelDisplay(data.stationId, data.value, data.timestamp);
});

evtSource.addEventListener('alert', function(event) {
    const data = JSON.parse(event.data);
    console.log('警报事件:', data);
    
    // 显示警报
    showAlertNotification(data);
});

// 连接打开
evtSource.addEventListener('open', function(event) {
    console.log('SSE连接已建立');
    updateConnectionStatus('connected');
});

// 错误处理
evtSource.addEventListener('error', function(event) {
    console.error('SSE错误');
    updateConnectionStatus('disconnected');
    
    if (event.target.readyState === EventSource.CLOSED) {
        console.log('连接已关闭');
    }
});

// 关闭连接
function closeEventStream() {
    evtSource.close();
    console.log('SSE连接已手动关闭');
}
\end{lstlisting}
